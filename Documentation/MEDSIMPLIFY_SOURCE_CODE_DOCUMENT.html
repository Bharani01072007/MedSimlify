1. System Overview & Architecture Flow

MedSimplify is a medical-grade, artificial intelligence-powered medical report simplification and healthcare management platform. It transforms complex diagnostic reports (CBC, Dengue profiles, LFT, Lipid profiles) into plain-language explanations with dual-tier summaries (Standard Medical + ELI5 Analogy) and multi-lingual regional Indian translations (Hindi, Tamil, Telugu).

Layer Technology Responsibility
Frontend SPA React 19, TypeScript, TanStack Router, TailwindCSS Mobile-frame responsive interface for patients and doctors.
API Backend Python 3.10+, FastAPI, Uvicorn Asynchronous REST API, CORS middleware, JWT security.
AI/NLP Pipeline PyMuPDF, PyTesseract, OpenCV, Helsinki-NLP Multi-modal OCR, entity extraction, simplification & translation.
Database PostgreSQL (Supabase) + SQLite Fallback Persistent relational storage with dynamic sequence realignment.

from sqlalchemy import create_engine, text
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
import os
from dotenv import load_dotenv

load_dotenv()

backend_dir = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
sqlite_fallback_url = f"sqlite:///{os.path.join(backend_dir, 'medsimplify.db').replace('\\', '/')}"
PRIMARY_DATABASE_URL = os.getenv("DATABASE_URL", sqlite_fallback_url)

if PRIMARY_DATABASE_URL and PRIMARY_DATABASE_URL.startswith("postgres://"):
    PRIMARY_DATABASE_URL = PRIMARY_DATABASE_URL.replace("postgres://", "postgresql://", 1)

try:
    engine = create_engine(PRIMARY_DATABASE_URL, pool_size=10, max_overflow=20, pool_pre_ping=True)
    with engine.connect() as conn:
        pass
    DATABASE_URL = PRIMARY_DATABASE_URL
except Exception:
    DATABASE_URL = sqlite_fallback_url
    engine = create_engine(DATABASE_URL, connect_args={"check_same_thread": False})

SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)
Base = declarative_base()

def get_db():
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

def sync_db_sequences(db):
    """Aligns PostgreSQL primary key serial sequences with existing table max(id)."""
    if "postgresql" in DATABASE_URL:
        tables = [
            "users", "patient_profiles", "doctor_profiles", "family_members",
            "chats", "messages", "medical_reports", "medicine_reminders",
            "medicine_logs", "appointments", "symptoms", "prescriptions", "medicines"
        ]
        for tbl in tables:
            try:
                db.execute(text(f"SELECT setval(pg_get_serial_sequence('{tbl}', 'id'), COALESCE((SELECT MAX(id) FROM {tbl}), 1));"))
                db.commit()
            except Exception:
                db.rollback()

4. Artificial Intelligence & Clinical NLP Pipeline

4.1 Multi-Modal OCR Text Extraction Service

backend/services/ocr_service.py
import io
import pytesseract
from PIL import Image
import cv2
import numpy as np

class OCRService:
    def __init__(self):
        self.custom_config = r'--oem 3 --psm 6'

    def preprocess_image(self, image_bytes: bytes):
        """Enhances contrast and performs Otsu thresholding."""
        try:
            image = Image.open(io.BytesIO(image_bytes))
            img_array = np.array(image)
            if len(img_array.shape) == 3 and img_array.shape[2] == 4:
                img_array = cv2.cvtColor(img_array, cv2.COLOR_RGBA2RGB)
            gray = cv2.cvtColor(img_array, cv2.COLOR_RGB2GRAY)
            gray = cv2.convertScaleAbs(gray, alpha=1.5, beta=30)
            gray = cv2.medianBlur(gray, 3)
            _, gray = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
            return gray
        except Exception:
            return None

    def extract_text(self, image_bytes: bytes, is_pdf: bool = False, filename: str = "") -> str:
        fname = filename.lower()
        if is_pdf or fname.endswith('.pdf'):
            import fitz
            doc = fitz.open(stream=image_bytes, filetype="pdf")
            return "\n".join([page.get_text() for page in doc if page.get_text()])
        
        preprocessed = self.preprocess_image(image_bytes)
        if preprocessed is not None:
            return pytesseract.image_to_string(preprocessed, config=self.custom_config)
        image = Image.open(io.BytesIO(image_bytes))
        return pytesseract.image_to_string(image, config=self.custom_config)

5.2 Medical Report Upload & Parsing Controller

backend/app/api/reports.py
from fastapi import APIRouter, Depends, HTTPException, UploadFile, File, Form
from sqlalchemy.orm import Session
from app.db.database import get_db
from app.models.models import MedicalReport, PatientProfile
from services.ai_pipeline import ai_pipeline

router = APIRouter(prefix="/reports", tags=["Medical Reports"])

@router.post("/upload")
async def upload_report(
    patient_id: int = Form(1),
    report_type: str = Form("lab_report"),
    language: str = Form("en"),
    file: UploadFile = File(...),
    db: Session = Depends(get_db)
):
    try:
        file_bytes = await file.read()
        is_pdf = file.filename.lower().endswith(".pdf")
        text = ai_pipeline.ocr_service.extract_text(file_bytes, is_pdf=is_pdf, filename=file.filename)
        
        output = ai_pipeline.process_report(text=text, language=language, filename=file.filename)
        
        report = MedicalReport(
            patient_id=patient_id,
            user_id=1,
            report_type=report_type,
            file_url=f"/reports/{file.filename}",
            file_name=file.filename,
            raw_text=text,
            extracted_data=output.get("extracted_data"),
            simplified_summary=output.get("simplified_summary"),
            translated_summary=output.get("translated_summary"),
            language=language,
            is_flagged=output.get("is_flagged", False)
        )
        db.add(report)
        db.commit()
        db.refresh(report)
        return report
    except Exception as e:
        db.rollback()
        raise HTTPException(status_code=500, detail=str(e))

6.1 Centralized Asynchronous API Client

frontend/src/lib/api-client.ts
const API_BASE_URL = typeof window !== "undefined" && window.location.hostname 
  ? `http://${window.location.hostname}:8000/api` 
  : "http://localhost:8000/api";

async function fetchApi<T>(endpoint: string, options: RequestInit = {}): Promise<T> {
  const url = `${API_BASE_URL}${endpoint}`;
  const defaultHeaders: Record<string, string> = {};

  if (!(options.body instanceof FormData)) {
    defaultHeaders["Content-Type"] = "application/json";
  }

  const token = typeof localStorage !== "undefined" ? localStorage.getItem("medsimplify_token") : null;
  if (token) {
    defaultHeaders["Authorization"] = `Bearer ${token}`;
  }

  const response = await fetch(url, { ...options, headers: { ...defaultHeaders, ...options.headers } });
  if (!response.ok) {
    throw new Error(`API error: ${response.status}`);
  }
  return response.json();
}

export const uploadReportApi = (data: FormData) => fetchApi("/reports/upload", { method: "POST", body: data });
export const getReportsApi = (id: number = 1) => fetchApi(`/reports/?patient_id=${id}`);
export const askAiAssistantApi = (q: string, ids: number[] = []) => fetchApi("/chat/ai/ask", { method: "POST", body: JSON.stringify({ question: q, report_ids: ids }) });
